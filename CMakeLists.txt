cmake_minimum_required(VERSION 3.16)
project(cpp_radix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_radix tests/test_radix.cpp)
target_include_directories(test_radix PRIVATE include)
if(MSVC)
  target_compile_options(test_radix PRIVATE /W4)
else()
  target_compile_options(test_radix PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME test_radix COMMAND test_radix)
